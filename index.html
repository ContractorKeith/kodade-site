// src/retry.ts

export function retry(fn, opts) {

  const max = opts.max ?? 3;

  let delay = opts.base ?? 250;

  for (let i = 0; i < max; i++) {

    try { return await fn(); }

    catch (e) { await sleep(delay); }

  }

}
